Оптимизируй SQL-запрос
SELECT arrayJoin(map(x -> x.name, x -> x.type, groupArray(system.columns))) AS columns
FROM system.columns
WHERE database = 'ym_sanok' AND table = 'visits_complete'
FORMAT JSON
Ваш текущий запрос получает структуру столбцов таблицы visits_complete из системной таблицы system.columns в ClickHouse, используя arrayJoin и map. Однако он не оптимален по нескольким причинам:

groupArray(system.columns) — пытается сгруппировать всю строку как структуру, что не поддерживается напрямую.
map(x -> x.name, x -> x.type, ...) — некорректный синтаксис: map() в ClickHouse создаёт словарь из двух массивов, но вы передаёте туда лямбды.
Запрос может завершиться ошибкой или дать непредсказуемые результаты.
✅ Цель

Если вы хотите получить список имён и типов столбцов таблицы visits_complete в удобном формате (например, для анализа схемы), вот оптимизированный и рабочий вариант:

SELECT 
    name AS column_name,
    type AS data_type
FROM system.columns
WHERE database = 'ym_sanok' AND table = 'visits_complete'
ORDER BY position;

def generate_clickhouse_sql(question: str, context: dict) -> str:
    # Здесь мог бы быть вызов ИИ (например, меня — Yandex WebSQL Concierge)
    # Но пока — простая логика на примере

    if "топ-10 городов по числу визитов за вчера" in question.lower():
        return """
SELECT RegionCity, COUNT(*) AS visit_count
FROM visits_complete
WHERE StartDate = today() - 1
GROUP BY RegionCity
ORDER BY visit_count DESC
LIMIT 10;
""".strip()

    if "столбцы таблицы visits_complete" in question.lower():
        return """
SELECT name, type
FROM system.columns
WHERE database = 'ym_sanok' AND table = 'visits_complete'
ORDER BY position;
""".strip()

    return "-- Не удалось распознать запрос. Уточните, пожалуйста."

3. main.py — API на FastAPI

from fastapi import FastAPI
from pydantic import BaseModel
from typing import Dict
import json

from schema import DatabaseContext
from assistant import generate_clickhouse_sql

app = FastAPI(title="SQL Assistant API")

class Request(BaseModel):
    question: str
    context: DatabaseContext

class Response(BaseModel):
    query: str
    engine: str
    database: str

@app.post("/generate-sql", response_model=Response)
async def generate_sql(request: Request):
    # В реальности здесь может быть вызов ИИ-модели
    sql_query = generate_clickhouse_sql(request.question, request.context.dict())
    
    return {
        "query": sql_query,
        "engine": request.context.databaseType,
        "database": request.context.databaseName
    }

# Запуск: uvicorn main:app --reload
